  { $match: { estado: "activo" }},
  { $lookup: {
      from: "turnos",
      localField: "_id",
      foreignField: "servicioId",
      as: "agenda"
  }},
  { $group: {
      _id: "$categoria",
      ventas: { $sum: "$total" }
